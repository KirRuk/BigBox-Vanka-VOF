cmake_minimum_required(VERSION 3.21)
project(SolverRegression LANGUAGES CXX)
enable_testing()
find_package(OpenMP REQUIRED COMPONENTS CXX)
find_package(MKL REQUIRED)
set(SOLVER_DIR "${CMAKE_CURRENT_LIST_DIR}/../Vanka_Feldman_patch_solver")
file(GLOB SOLVER_SOURCES CONFIGURE_DEPENDS "${SOLVER_DIR}/src/*.cpp")
list(FILTER SOLVER_SOURCES EXCLUDE REGEX "(Mumps_Solver|Vanka_Feldman_patch_solver)\\.cpp$")
add_executable(memory_regression memory_regression.cpp periodic_conservation.cpp ${SOLVER_SOURCES})
target_include_directories(memory_regression PRIVATE "${SOLVER_DIR}/include")
target_compile_features(memory_regression PRIVATE cxx_std_20)
target_link_libraries(memory_regression PRIVATE MKL::MKL OpenMP::OpenMP_CXX)
add_test(NAME memory_regression COMMAND memory_regression)
set_tests_properties(memory_regression PROPERTIES RUN_SERIAL TRUE)
